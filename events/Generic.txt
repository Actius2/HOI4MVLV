# Notification event that leader has gotten sick

	id = generic.17
	title = generic.17.t
	desc = generic.17.d

		hidden_effect = {
			FROM = {
				set_unit_leader_flag = {

					value = 1
					days = 180
				}
			}
		}
	}

		name = unfortunate
		effect_tooltip = {
			FROM = {
				add_timed_unit_leader_trait = {
					trait = sick
					days = 30
				}
			}
		}
	}	
}
