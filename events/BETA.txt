##横浜基地襲撃BETA
country_event = {
	id = BETA.11
	desc = BETA.11.d
	title = BETA.11.t
	picture = BETA_event_1.dds
	is_triggered_only = yes
	immediate = {
		hidden_effect = {
			if = {
				limit = {
					NOT = { has_global_flag = H21_broken_flag }
				}
				if = {
					#H21が所有していないならば
					limit = {
						NOT = {
							H21 = { controls_state = 789 }
						}
					}
					H21 = {
						set_province_controller = 13218
						load_oob = H21_yokohama
						add_timed_idea = {
							idea = H21_sado
							days = 50
						}
						add_timed_idea = {
							idea = mvlv_aggressive_beta
							days = 50
						}
					}
					JAP = {
						country_event = mvlv_uni.21
						play_song_Alternative_41_effect = yes
					}
					UNI = {
						country_event = mvlv_uni.21
						play_song_Alternative_41_effect = yes
					}
					set_global_flag = H21_broken_flag
				}
				#予備判定
				BET = {
					country_event = {
						id = BETA.11
						days = 4
					}
				}
			}
		}
	}
	option = {
		name = BETA.11.a
	}
}

######佐渡島イベント
#佐渡島におけるBETA噴出
country_event = {
	id = BETA.12
	desc = BETA.12.d
	title = BETA.12.t
	picture = BETA_event_1.dds
	is_triggered_only = yes
	immediate = {
		hidden_effect = {
			###通知
			####場合によっては空振り通知になる可能性あり
			##しばらく保留で問題になるようなら噴出処理内に

			####オペ佐渡時の動作
			if = {
				limit = {
					has_global_flag = mvlv_op_sado_flag
				}

				##緊急避難のイベント
				every_country = {
					limit = {
						is_in_faction_with = UNI
					}
					news_event = { id = mvlv_news.8 days = 1 }
				}
			}
			##噴出処理
			H21 = {
				if = {
					limit = {
						NOT = { has_country_flag = mvlv_gushing_beta_flag }
					}
					every_country = {
						limit = {
							is_in_faction_with = UNI
						}
						country_event = mvlv_uni.40
					}
					H21 = {
						#set_province_controller = 9952
						load_oob = H21_empop
						add_timed_idea = {
							idea = MVLV_BETA_idea_gushing
							days = 2
						}
					}
					set_country_flag = mvlv_gushing_beta_flag
				}
				###横浜襲撃(初回判定)＞予備判定はBETA11で
			}
			if = {
				limit = {
					NOT = { has_global_flag = H21_broken_flag }
				}
				BET = {
					country_event = {
						id = BETA.11
						days = 4
					}
				}
			}
		}
	}
	option = {
		name = BETA.12.a
	}
}





#ハイヴ奪取時ハイヴ占領アイデア付与、取り返されたら時限剥奪
country_event = {
	id = BETA.98
	desc = BETA.98.d
	title = BETA.98.t
	is_triggered_only = yes
	option = {
		name = BETA.98.a
		if = {
			limit = {
				NOT = {
					has_idea = mvlv_brokenhive
				}
			}
			add_ideas = {
				mvlv_brokenhive
			}
			if = {
				limit = {
					NOT = {
						tag = H00
					}
					NOT = {
						tag = H21
					}
				}
				news_event = {
					id = mvlv_news.1
					hours = 1
				}
			}
			#佐渡陥落
			else_if = {
				limit = {
					tag = H21
				}
				country_event = { id = BETA.12 days = 1 }
				every_country = {
					limit = { has_template = "A-02" }
					delete_unit_template_and_units = {
						division_template = "A-02"
					}
				}
				news_event = mvlv_news.1
				##old#news_event = mvlv_news.3
			}
			else_if = {
				limit = {
					tag = H00
				}
				BET = {
					add_ideas = {
						mvlv_brokenhive
					}
				}
				if = {
					limit = {
						AND = {
							NOT = {
								has_idea = BET_broken_original_hive
							}
							NOT = {
								has_idea = BET_end_game
							}
						}
					}
					BET = {
						add_to_faction = H00
					}
					every_country = {
						limit = {
							is_in_faction_with = UNI
						}
						country_event = mvlv_uni.7
					}
					every_country = {
						limit = {
							OR = {
								is_in_faction_with = BET
								tag = H00
							}
						}
						country_event = BETA.7
					}
					else = {
						news_event = {
							id = mvlv_news.1
							hours = 1
						}
					}
				}
			}
		}
		else_if = {
			limit = {
				has_idea = mvlv_brokenhive
				NOT = { tag = H21 }
			}
			country_event = {
				id = BETA.99
				days = 10
			}
		}
		else_if = {
			limit = {
				tag = H21
				NOT = {
					has_global_flag = mvlv_op_sado_flag
				}
			}
			country_event = {
				id = BETA.99
				days = 10
			}
		}
	}
}

				}

					hours = 1
				}
				else = {
					BET = {
						remove_ideas = {
							mvlv_brokenhive
						}
					}
					news_event = {
						id = mvlv_news.2
						hours = 1
					}
				}
			}
		}
	}
