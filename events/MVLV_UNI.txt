		name = mvlv_uni.12.a

			}
		}

			factor = 100
		}
		if = {
			limit = {
				has_equipment = {
					MVLV_TSF_chassis < 250
				}
			}
			country_event = {
				id = mvlv_uni.12
				days = 3
			}
			custom_effect_tooltip = MVLV_event_xmos_tt
			else = {
				send_equipment = {
					equipment = MVLV_TSF_chassis
					amount = 250
					target = UNI
				}
				add_ideas = UNI_xmosi
				set_technology = {
					MVLV_research_OS_XM3 = 1
					MVLV_research_OS_close_XM3 = 1
					MVLV_research_OS_support_XM3 = 1
				}
				UNI = {
					country_event = mvlv_uni.13
				}
			}
		}
	}

		name = mvlv_uni.12.b
		UNI = {
			country_event = mvlv_uni.13

				type = arms_factory

		}

		}
