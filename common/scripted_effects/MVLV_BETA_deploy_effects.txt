		}
		create_unit = {
			allow_spawning_on_enemy_provs = yes
			division = "name = \"Destroyer class division\" division_template = \"Destroyer class division\" start_experience_factor = 0.5"
			owner = PREV
			count = 2
		}
		PREV = {
			if = {
				limit = {
					has_equipment = {
						motorized_equipment_1 < 2000
					}
				}
				add_equipment_to_stockpile = {
					type = motorized_equipment_1
					amount = 1000
					producer = PREV
				}
			}
		}


		if = {
			limit = {
				PREV = { check_variable  = { BETA_deploy_count > 6 } }	
			}
			create_unit = {
				allow_spawning_on_enemy_provs = yes
				division = "name = \"Fort class division\" division_template = \"Fort class division\" start_experience_factor = 0.5"
				owner = PREV
				count = 1
			}
			if = {
				limit = {
					#特定時期でない場合レーザー補給
					OR = {
						NOT = { has_global_flag = MVLV_underground_flag }
						NOT = { has_global_flag = mvlv_alternative_hell_news_flag }
						has_global_flag = mvlv_alternative_hell_laser_flag
					}
				}
				PREV = { 
					add_equipment_to_stockpile = {
						type = laser_equipment
						amount = 500
						producer = PREV
					}
					add_equipment_to_stockpile = {
						type = heavy_laser_equipment
						amount = 100
						producer = PREV
					}
					set_variable = { BETA_deploy_count = 0 }
				}
			}
		}

	}
