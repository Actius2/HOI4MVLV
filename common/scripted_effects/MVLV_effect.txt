MVLV_AI_bunker = {
	every_neighbor_country = {
		limit = {
			is_in_faction_with = UNI
			NOT = { tag = JAP }
			is_ai = yes
		}
		#日本の前線で正しく要塞が建築される方
		#every_neighbor_state
		every_controlled_state = {
			limit = {
				is_controlled_by = PREV
				num_owned_neighbour_states ={
					owner = ROOT
					count > 0
				}
			}
			set_building_level = {
				type = bunker
				level = 10
				instant_build = yes
				province = {
					all_provinces = yes
					limit_to_border = yes
				}
			}
			set_building_level = {
				type = coastal_bunker
				level = 10
				instant_build = yes
				province = {
					all_provinces = yes
					limit_to_border = yes
					limit_to_coastal = yes
				}
			}
		}
	}
}

CAN_Towards_independence_scripte_effect = {

	if = {
		limit = {
			is_fully_controlled_by = ROOT
		}
		if = {
			limit = {
				industrial_complex < 2
			}
			ROOT = {
				add_offsite_building ={
					type = industrial_complex
					level = -2
				}
			}
		}
		else = {
			remove_building = {
				type = industrial_complex
				level = 2
			}
		}
		add_building_construction = {
			type = arms_factory 
			level = 2
			instant_build = yes
		}
	}
}

CAN_Atomic_down_script_effect = {
	ROOT = {
		if = {
			limit = {
				has_idea = CAN_atomic_5
			}
			swap_ideas = {
				remove_idea = CAN_atomic_5
				add_idea = CAN_atomic_4
			}
		}
		else_if = {
			limit = {
				has_idea = CAN_atomic_4
			}
			swap_ideas = {
				remove_idea = CAN_atomic_4
				add_idea = CAN_atomic_3
			}
		}
		else_if = {
			limit = {
				has_idea = CAN_atomic_3
			}
			swap_ideas = {
				remove_idea = CAN_atomic_3
				add_idea = CAN_atomic_2
			}
		}
		else_if = {
			limit = {
				has_idea = CAN_atomic_2
			}
			swap_ideas = {
				remove_idea = CAN_atomic_2
				add_idea = CAN_atomic_1
			}
		}
		else_if = {
			limit = {
				has_idea = CAN_atomic_1
			}
			swap_ideas = {
				remove_idea = CAN_atomic_1
				add_idea = CAN_atomic
			}
		}
	}
}
