on_actions = {
	on_startup = {
		effect = {
			every_country = {
				#add_ideas = test_food_ideas
			}
			every_country = {
				limit = {
					is_in_faction_with = UNI
				}
				if = {
					limit = {
						NOT = {
							has_dynamic_modifier = food_effects
						}
					}
					add_dynamic_modifier = {
						modifier = food_effects
					}
				}
				#set_technology = { synth_oil_experiments = 1 }
				
				set_variable = {
					var = country_manpower
					value = max_manpower_k 
				}
				set_variable = {
					var = country_army_manpower
					value = deployed_army_manpower_k 
				}
				set_variable = {
					var = country_navy_manpower
					value = deployed_navy_manpower_k 
				}
				set_variable = {
					var = country_airforce_manpower
					value = deployed_airforce_manpower_k 
				}
				set_variable = {
					var = need_food
					value = 0
				}
				set_variable = {
					var = need_food_military
					value = 0
				}
				set_variable = {
					var = eat_food
					value = 0
				}

				multiply_variable = {
					var = country_manpower
					value = 0.001
				}
				add_to_variable = { 
					var = need_food 
					value = country_manpower 
				}
				multiply_variable = {
					var = country_army_manpower
					value = 0.1
				}
				add_to_variable = {
					var = need_food_military
					value = country_army_manpower
				}
				multiply_variable = {
					var = country_navy_manpower
					value = 0.1
				}
				add_to_variable = {
					var = need_food_military
					value = country_navy_manpower
				}
				multiply_variable = {
					var = country_airforce_manpower
					value = 0.1
				}
				add_to_variable = {
					var = need_food_military
					value = country_airforce_manpower
				}
				if = {
					limit = {
						has_idea = stop_rationing
					}
					set_variable = {
						var = need_food
						value = 0
					}
				}
				if = {
					limit = {
						has_idea = subsistence_rationing
					}
					multiply_variable = {
						var = need_food
						value = 0.5
					}
				}
				if = {
					limit = {
						has_idea = war_rationing
					}
					multiply_variable = {
						var = need_food
						value = 0.75
					}
				}
				if = {
					limit = {
						has_idea = no_rationing
					}
					multiply_variable = {
						var = need_food
						value = 1.35
					}
				}

				subtract_from_variable = {
					eat_food = need_food
				}
				subtract_from_variable = {
					eat_food = need_food_military
				}
			}
		}
	}
