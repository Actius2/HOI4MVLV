				IF = {

				if = {

			467 = {
				set_state_flag = CAN_nuclear_explosion_flag

	on_state_control_changed = {
		effect = {

		
			#target:元の所有国がBETA陣営
			#effect:BETA陣営で降伏値が上がったら陥落アイデア付与
			#________________________________________________________
			#target:今の所有国がBETA陣営
			#effect:BETA陣営で降伏値が下がっていれば陥落アイデア剥奪
			if = {
				limit = {
					FROM = {
						OR = {
							is_in_faction_with = BET
							tag = H00
						}
						surrender_progress > 0.3
						not = {
							has_idea = mvlv_brokenhive
						}
					}
				}
				FROM = {
					country_event = BETA.98
				}
			}
			else_if = {
				limit = {
					OR = {
						is_in_faction_with = BET
						tag = H00
					}
					surrender_progress < 0.1
					has_idea = mvlv_brokenhive
				}
				country_event = BETA.98
			}
			
			#########保険
			if = {
				limit = {
					H00 = {
						NOT = { is_in_faction_with = BET }
						num_divisions < 1
					}
				}
				BET = { add_to_faction = H00 }
				UNI = {
					annex_country = {
						target = H00
						transfer_troops = no
					}
				}
			}
		}
	}
	on_capitulation_immediate = {
		#########保険
		effect = {
			if = {
				limit = {
					H00 = {
						NOT = { is_in_faction_with = BET }
						num_divisions < 1
					}
				}
				BET = { add_to_faction = H00 }
				UNI = {
					annex_country = {
						target = H00
						transfer_troops = no
					}
				}
			}
		}
	}
	on_capitulation = {
		effect = {
			#target:降伏したBETA陣営で発生
			#effect:BETA陣営で降伏値が上がったら陥落アイデア付与
			#effect:BETA陣営で降伏値が下がっていれば陥落アイデア剥奪
			if = {
				limit = {
					OR = {
						is_in_faction_with = BET
						tag = H00
					}
					not = {
						has_idea = mvlv_brokenhive
					}
				}
				country_event = BETA.98
			}
		}
	}
	on_uncapitulation = {
		effect = {
			#target:降伏したBETA陣営で発生
			#effect:BETA陣営で降伏値が下がっていれば陥落アイデア剥奪
			if = {
				limit = {
					OR = {
						is_in_faction_with = BET
						tag = H00
					}
					has_idea = mvlv_brokenhive
				}
				country_event = BETA.98
			}
		}
	}
	on_nuke_drop = {
		effect = {
			if = {
				limit = {
					has_global_flag = mvlv_sado_G_flag
					FROM = { state = 789 }
				}
				H21 = { transfer_state = 801 }
				SRD = {
					transfer_state = 789
					#set_state_owner = 789
					#set_state_controller = 789
					#set_province_name = { id = 789 name = LOC_KEY }
					division_template = {
						name = "Infantry Division"
						is_locked = yes
						division_cap = 3 
						division_names_group = USA_INF_01
						priority = 0
						template_counter = 0
						regiments = {
							infantry = { x = 0 y = 0 }
							infantry = { x = 0 y = 1 }
							infantry = { x = 0 y = 2 }
							infantry = { x = 0 y = 3 }
						}
					}
				}
				789 = {
					create_unit = {
						division = "name = \"Infantry Division\" division_template = \"Infantry Division\" start_experience_factor = 0.5"
						owner = SRD
						allow_spawning_on_enemy_provs = yes
					}
				}
				SRD = {
					delete_unit = {
						state = 789
						disband = no
					}
				}
			}
			if = {
				limit = {
					ROOT = {
						NOT = {
							OR = {
								is_in_faction_with = BET
								tag = SRD
								has_global_flag = mvlv_sado_G_flag
							}
						}
					}
				}
				##BETAかSRD以外が落とした場合に発生
				add_to_variable = { global.nuke_drop_num = 1 }
				if = {
					limit = {
						check_variable = { global.nuke_drop_num > 0 }
					}
					if = {
						limit = {
							check_variable = { global.nuke_drop_num = 1 }
						}
						every_country = {
							limit = {
								is_in_faction_with = UNI
							}
							set_variable = { global.MVLV_effect_G_Lancher_modifier_value = -0.05 }
							add_dynamic_modifier = { modifier = MVLV_effect_G_Lancher_modifier }
						}
					}
					if = {
						limit = {
							check_variable = { global.nuke_drop_num = 2 }
						}
						subtract_from_variable = { global.MVLV_effect_G_Lancher_modifier_value = 0.05 }
					}
					if = {
						limit = {
							check_variable = { global.nuke_drop_num > 2 }
						}
						subtract_from_variable = { global.MVLV_effect_G_Lancher_modifier_value = 0.1 }
					}
					if = {
						limit = {
							check_variable = { global.nuke_drop_num = 7 }
						}
						every_country = {
							limit = {
								is_in_faction_with = UNI
							}
							country_event = mvlv_generic.38
						}
					}
					FROM = {
						#ニュース
						every_country = {
							if = {
								limit = {
									PREV = { is_core_of = THIS }
									tag = ROOT
								}
								country_event = mvlv_generic.37
							}
							if = {
								limit = {
									controls_state = PREV
									OR = {
										tag = BET
										tag = H00
									}
								}
								#BETAなら自身のBETAPOP
								BET = {
									MVLV_BETA_start_deploy100_effect = yes
								}
							}
							if = {
								limit = {
									is_in_faction_with = BET
									owns_state = PREV
									PREV = {
										NOT = { is_core_of = THIS }
									}
								}
								#BETAなら自身のBETAPOP
								MVLV_BETA_start_deploy100_effect = yes
							}
							if = {
								limit = {
									is_in_faction_with = BET
									owns_state = PREV
									PREV = { is_core_of = THIS }
								}
								#BETAなら周囲のBETAPOP
								#一旦保留
							}
							if = {
								limit = {
									is_in_faction_with = UNI
									NOT = { has_full_control_of_state = PREV }
									PREV = { is_core_of = PREV }
									NOT = { tag = ROOT }
								}
								ROOT = {
									PREV = {
										#event
										country_event = mvlv_generic.34
									}
								}
							}			
						}
					}
				}
			}
		}
	}
