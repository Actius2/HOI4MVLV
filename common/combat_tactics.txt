tactic_counterattack = {
	# counters basic attack, assault
	is_attacker = no
	trigger = {
		is_attacker = no
		skill_advantage > 0
		phase = no
		NOT = {
			is_fighting_in_terrain = hive_mainhall
			is_fighting_in_terrain = hive_gate
			is_fighting_in_terrain = hive_gateclass
			is_fighting_in_terrain = hive_drift
			is_fighting_in_terrain = hive_super_mainhall
		}
	}
	active = yes
	base = {
		factor = 4
		modifier = {
			add = 4
			has_trait = unyielding_defender
		}
	}
	picture = counterattack
	defender = 0.25
}
